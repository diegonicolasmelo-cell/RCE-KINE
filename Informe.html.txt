function drawDonut(svgId,legId,items){
  const svg=$(svgId),leg=$(legId);
  if(!svg||!leg) return;
  const R=48,C=2*Math.PI*R,total=items.reduce((a,b)=>a+(b.v||0),0);
  const fil=items.filter(i=>i.v>0);
  let off=0,segs='';
  fil.forEach(i=>{
    const len=(i.v/total)*C;
    segs+=`<circle r="${R}" cx="70" cy="70" fill="none" stroke="${i.c}" stroke-width="18"
      stroke-dasharray="${len} ${C-len}" stroke-dashoffset="${-off}" transform="rotate(-90 70 70)"/>`;
    off+=len;
  });
  svg.innerHTML=segs+`<text x="70" y="66" text-anchor="middle" font-size="22" font-weight="800" fill="#1d1d1f">${total}</text>
    <text x="70" y="81" text-anchor="middle" font-size="9" fill="#8e8e93" font-weight="600">TOTAL</text>`;
  leg.innerHTML=fil.map(i=>`<div class="dleg">
    <span class="dleg-dot" style="background:${i.c}"></span>
    <span class="dleg-lbl">${i.lbl}</span>
    <span class="dleg-pct">${i.v} · ${pct(i.v,total)}</span></div>`).join('');
}

function getLight(val,thresholds){
  // thresholds: {greenLow:x,yellowLow:x,higher_is_better:bool}
  const {lo,hi,better}=thresholds;
  if(better==='high') return val>=hi?'green':val>=lo?'yellow':'red';
  if(better==='low')  return val<=lo?'green':val<=hi?'yellow':'red';
  return 'gray';
}

    {lbl:'Autoextubaciones',sub:'n casos',k24:d24.vm_auto,k25:d25.vm_auto,better:'low',raw24:d24.vm_auto,raw25:d25.vm_auto},
    {lbl:'Extubaciones programadas',sub:'total',k24:d24.ep_total,k25:d25.ep_total,better:'neutral'},
    {lbl:'Éxito extubación progr.',sub:'tasa',k24:pct(d24.ep_exitosas,d24.ep_total),k25:pct(d25.ep_exitosas,d25.ep_total),better:'high',raw24:pctN(d24.ep_exitosas,d24.ep_total),raw25:pctN(d25.ep_exitosas,d25.ep_total)},
    {lbl:'Reintubaciones',sub:'n casos',k24:d24.ri_total,k25:d25.ri_total,better:'low',raw24:d24.ri_total,raw25:d25.ri_total},
    {lbl:'Tasa de reintubación',sub:'% de extubaciones',k24:pct(d24.ri_total,d24.ep_total),k25:pct(d25.ri_total,d25.ep_total),better:'low',raw24:pctN(d24.ri_total,d24.ep_total),raw25:pctN(d25.ri_total,d25.ep_total)},
    {lbl:'Causa 1ª sin protocolo',sub:'motivo',k24:'Agitación psicomotora',k25:'Agitación psicomotora',better:'neutral'},
    {lbl:'Causa 1ª reintubación',sub:'motivo',k24:'Falla resp. post-ext.',k25:'Falla resp. / Mal manejo secrec.',better:'neutral'},
    {lbl:'TQT total',sub:'traqueostomías año',k24:d24.tqt_total||'—',k25:d25.tqt_total,better:'neutral'},
    {lbl:'TQT primeros 10 días',sub:'% del total',k24:d24.tqt_total?pct((d24.tqt_d1a5||0)+(d24.tqt_d6a10||0),d24.tqt_total):'—',k25:pct(d25.tqt_d1a5+d25.tqt_d6a10,d25.tqt_total),better:'high',raw24:d24.tqt_total?pctN((d24.tqt_d1a5||0)+(d24.tqt_d6a10||0),d24.tqt_total):null,raw25:pctN(d25.tqt_d1a5+d25.tqt_d6a10,d25.tqt_total)}
  ];

  function trend(r){
    if(r.better==='neutral'||r.raw24==null||r.raw25==null) return '<span class="trend-neutral">—</span>';
    const diff=r.raw25-r.raw24;
    if(Math.abs(diff)<1) return '<span class="trend-neutral">→ Sin cambio</span>';
    const improved=(r.better==='high'&&diff>0)||(r.better==='low'&&diff<0);
    const arrow=diff>0?'↑':'↓';
    const cls=improved?'trend-up':'trend-down';
    return `<span class="${cls}">${arrow} ${improved?'Mejora':'Empeora'}</span>`;
  }

  $('cmpTable').innerHTML=`<thead><tr>
    <th>Indicador</th><th>2024</th><th>2025</th><th>Tendencia</th>
  </tr></thead><tbody>`+rows.map(r=>`<tr>
    <td><span class="lbl">${r.lbl}</span><br><span class="sub">${r.sub}</span></td>
    <td>${r.k24}</td><td>${r.k25}</td><td>${trend(r)}</td>
  </tr>`).join('')+'</tbody>';
}

// ═══════════════ INPUTS ═══════════════
function bindInputs(){
  document.querySelectorAll('.ei[data-key]').forEach(inp=>{
    const k=inp.dataset.key;
    inp.value=DATA[k]!==undefined?DATA[k]:'';
    inp.oninput=()=>{
      DATA[k]=parseInt(inp.value)||0;
      document.querySelectorAll(`.ei[data-key="${k}"]`).forEach(el=>{if(el!==inp)el.value=DATA[k];});
      renderAll(); saveData();
    };
  });
  document.querySelectorAll('.ei-text[data-key]').forEach(inp=>{
    const k=inp.dataset.key;
    inp.value=DATA[k]||'';
    inp.oninput=()=>{ DATA[k]=inp.value; renderAll(); saveData(); };
  });
  document.querySelectorAll('[data-note]').forEach(ta=>{
    const k='note_'+ta.dataset.note;
    ta.value=DATA[k]||'';
    ta.oninput=()=>{ DATA[k]=ta.value; saveData(); };
  });
  const resp=$('m-resp');
  if(resp){ resp.value=DATA.responsable||''; resp.oninput=()=>{ DATA.responsable=resp.value; saveData(); }; }
}

function loadData(yr){
  try{ const s=localStorage.getItem('kine_est_'+yr); return s?JSON.parse(s):JSON.parse(JSON.stringify(DEFAULTS[yr])); }
  catch(e){ return JSON.parse(JSON.stringify(DEFAULTS[yr])); }
}
function saveData(){ try{ localStorage.setItem('kine_est_'+YR,JSON.stringify(DATA)); }catch(e){} }

function setYear(yr){
  if(COMPARE) return;
  YR=yr;
  ['2024','2025'].forEach(y=>$(('btn'+y))?.classList.toggle('on',y===yr));
  DATA=loadData(yr);
  bindInputs(); renderAll();
}

function toggleCompare(){
  COMPARE=!COMPARE;
  $('btnCmp').classList.toggle('on',COMPARE);
  $('cmpSection').style.display=COMPARE?'block':'none';
  if(COMPARE){ renderComparison(); }
}

function resetYear(){
  if(!confirm(`¿Restaurar datos de ${YR} al informe original?`)) return;
  localStorage.removeItem('kine_est_'+YR);
  DATA=JSON.parse(JSON.stringify(DEFAULTS[YR]));
  bindInputs(); renderAll();
}

}

function loadVis(){
  try{ const s=localStorage.getItem('kine_vis'); return s?{...VIS_DEFAULTS,...JSON.parse(s)}:{...VIS_DEFAULTS}; }
  catch(e){ return {...VIS_DEFAULTS}; }
}

function applyVisibility(){
  const map={
    'sec-1':VIS.sec1,'sec-1-ktr':VIS.sec1&&VIS.sec1_ktr,'sec-1-ktm':VIS.sec1&&VIS.sec1_ktm,
    'sec-2':VIS.sec2,'sec-2-auto':VIS.sec2&&VIS.sec2_auto,
    'sec-3':VIS.sec3,'sec-3-dn':VIS.sec3&&VIS.sec3_dn,'sec-3-exito':VIS.sec3&&VIS.sec3_exito,'sec-3-sprot':VIS.sec3&&VIS.sec3_sprot,
    'sec-4':VIS.sec4,'sec-5':VIS.sec5,'sec-5-mot':VIS.sec5&&VIS.sec5_mot,'sec-5-disp':VIS.sec5&&VIS.sec5_disp,
    'sec-6':VIS.sec6,'sec-6-dias':VIS.sec6&&VIS.sec6_dias,'sec-6-diag':VIS.sec6&&VIS.sec6_diag,
    'sec-7':VIS.sec7
  };
  Object.entries(map).forEach(([id,show])=>{
    const el=$(id); if(el) el.style.display=show?'':((id.startsWith('sec-')&&id.split('-').length===2)?'':'none');
  });
  // Notes
  document.querySelectorAll('.note-field').forEach(el=>{ el.style.display=VIS.showNotes?'':'none'; });
}
